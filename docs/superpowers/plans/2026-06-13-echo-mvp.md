# Echo MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-use "song-library safe + scene recommendation queue": a FastAPI+SQLite backend that snapshots a YouTube playlist daily and detects disappeared songs, plus an installable PWA that serves a 10–30 song scene-aware queue and logs feedback.

**Architecture:** Single FastAPI process hosts the JSON API, an APScheduler daily snapshot job, and the PWA static files. SQLite (single file) is the store. Deterministic core logic (snapshot diff, title normalization, recommendation scoring, feedback aggregation) is pure and unit-tested; YouTube I/O is isolated behind one module and mocked in tests. Playback happens by opening a YouTube Music link.

**Tech Stack:** Python 3.11, FastAPI, SQLAlchemy 2.0, SQLite, APScheduler, google-api-python-client + google-auth-oauthlib (YouTube Data API v3), pytest. Frontend: vanilla HTML/JS PWA (manifest + service worker).

```python
import re

_BRACKETS = re.compile(r"[\(\[].*?[\)\]]")
_FEAT = re.compile(r"\b(feat|ft|featuring)\.?\b.*", re.IGNORECASE)
_NONWORD = re.compile(r"[^\w\s]", re.UNICODE)
_WS = re.compile(r"\s+")


def normalize_title(title: str) -> str:
    s = title or ""
    s = _BRACKETS.sub(" ", s)
    s = _FEAT.sub(" ", s)
    s = _NONWORD.sub(" ", s)
    s = _WS.sub(" ", s).strip().lower()
    return s


def normalize_artist(artist: str) -> str:
    s = artist or ""
    s = re.sub(r"\s*-\s*topic\s*$", "", s, flags=re.IGNORECASE)
    s = _NONWORD.sub(" ", s)
    s = _WS.sub(" ", s).strip().lower()
    return s
```

```python
from sqlalchemy import select
from sqlalchemy.orm import Session
from echo.models import Track
from echo.text_norm import normalize_title, normalize_artist

DURATION_TOLERANCE = 8  # seconds


def upsert_track(session: Session, *, video_id: str, title: str,
                 artist: str = "", duration: int = 0) -> Track:
    track = session.scalar(select(Track).where(Track.youtube_video_id == video_id))
    if track is None:
        track = Track(youtube_video_id=video_id)
        session.add(track)
    track.title = title
    track.artist = artist
    track.duration_seconds = duration
    track.norm_title = normalize_title(title)
    track.norm_artist = normalize_artist(artist)
    return track


def find_replacements(session: Session, *, exclude_video_id: str, title: str,
                      artist: str, duration: int) -> list[Track]:
    nt, na = normalize_title(title), normalize_artist(artist)
    rows = session.scalars(
        select(Track).where(Track.norm_title == nt,
                            Track.youtube_video_id != exclude_video_id)
    ).all()
    out = []
    for r in rows:
        if na and r.norm_artist and r.norm_artist != na:
            continue
        if duration and r.duration_seconds and abs(r.duration_seconds - duration) > DURATION_TOLERANCE:
            continue
        out.append(r)
    return out
```

```python
from dataclasses import dataclass, field


@dataclass
class SnapItem:
    source_track_id: str
    title_at_time: str = ""
    artist_at_time: str = ""
    position: int = 0
    available: bool = True


@dataclass
class Renamed:
    source_track_id: str
    old_title: str
    new_title: str


@dataclass
class Changes:
    missing: list[SnapItem] = field(default_factory=list)
    added: list[SnapItem] = field(default_factory=list)
    renamed: list[Renamed] = field(default_factory=list)


def diff(prev: list[SnapItem], cur: list[SnapItem]) -> Changes:
    prev_by_id = {i.source_track_id: i for i in prev}
    cur_by_id = {i.source_track_id: i for i in cur}
    ch = Changes()
    for vid, item in prev_by_id.items():
        if vid not in cur_by_id:
            ch.missing.append(item)
    for vid, item in cur_by_id.items():
        if vid not in prev_by_id:
            ch.added.append(item)
        elif prev_by_id[vid].title_at_time != item.title_at_time:
            ch.renamed.append(Renamed(vid, prev_by_id[vid].title_at_time, item.title_at_time))
    ch.missing.sort(key=lambda i: i.source_track_id)
    ch.added.sort(key=lambda i: i.source_track_id)
    ch.renamed.sort(key=lambda r: r.source_track_id)
    return ch
```

```python
    return session.scalar(

        .order_by(PlaylistSnapshot.captured_at.desc())
    )


def run_watchdog(session: Session, playlist_id: str, fetched: list[SnapItem],
                 search_fn=None) -> Changes:
    """Capture a new snapshot, diff against previous OK snapshot, raise alerts for missing."""
    prev_snap = _latest_ok_snapshot(session, playlist_id)
    prev_items = [SnapItem(i.source_track_id, i.title_at_time, i.artist_at_time,
                           i.position, i.available) for i in (prev_snap.items if prev_snap else [])]
    new_snap = capture(session, playlist_id, fetched, ok=True)
    changes = diff(prev_items, fetched)
    for m in changes.missing:
        cands = [
            {"videoId": c.youtube_video_id, "title": c.title, "source": "registry"}
            for c in find_replacements(session, exclude_video_id=m.source_track_id,
                                       title=m.title_at_time, artist=m.artist_at_time, duration=0)
        ]
        if not cands and search_fn is not None:
            q = f"{m.title_at_time} {m.artist_at_time}".strip()
            cands = [{**c, "source": "youtube_search"} for c in search_fn(q)]
        session.add(Alert(
            snapshot_id=new_snap.id, kind="missing", source_track_id=m.source_track_id,
            title_at_time=m.title_at_time, artist_at_time=m.artist_at_time,
            candidate=json.dumps(cands, ensure_ascii=False),
```

```python
from datetime import datetime
from sqlalchemy import select
from sqlalchemy.orm import Session
from echo.models import Track, TrackContextStats, TrackTag

WEIGHTS = {
    "base_like": 1.0,
    "scene": 2.0,
    "time": 1.0,
    "completion": 1.5,
    "rediscovery": 1.0,
    "overplay": -2.0,
    "skip": -0.75,
    "seed": 3.0,
}

# "听腻了" mode: surface long-unplayed favorites. Scene/time/seed matching is
# dropped; staleness dominates and recently-played tracks are pushed down.
REDISCOVER_WEIGHTS = {**WEIGHTS, "scene": 0.0, "time": 0.0, "seed": 0.0,
                      "rediscovery": 5.0}
REDISCOVER_SCENE = "rediscover"


def _rediscovery(now: datetime, last) -> float:
    if last is None:
        return 0.5
    days = (now - last).days
    if days >= 14:
        return 1.0
    if days >= 7:
        return 0.5
    return 0.0


def _overplay(now: datetime, last) -> float:
    if last is None:
        return 0.0
    hours = (now - last).total_seconds() / 3600.0
    if hours < 24:
        return 1.0
    if hours < 48:
        return 0.5
    return 0.0


def score_track(now: datetime, scene_stat, time_stat, global_agg, seed_match: float,
                weights: dict | None = None) -> float:
    w = weights or WEIGHTS
    s = scene_stat
    tb = time_stat
    base_like = 0.5 * global_agg["manual_pick"] + 1.0 * global_agg["avg_completion"]
    scene_match = 1.0 * s["manual_pick"] + 0.5 * s["play"] - 1.0 * s["skip"]
    time_match = 0.5 * tb["play"] - 0.5 * tb["skip"]
    completion = s["avg_completion"]
    redis = _rediscovery(now, s["last"])
    over = _overplay(now, s["last"])
    skip_pen = s["skip"]
    return (
        w["base_like"] * base_like
        + w["scene"] * scene_match
        + w["time"] * time_match
        + w["completion"] * completion
        + w["rediscovery"] * redis
        + w["overplay"] * over
        + w["skip"] * skip_pen
        + w["seed"] * seed_match
    )


def _empty_stat():
    return {"play": 0, "skip": 0, "manual_pick": 0, "avg_completion": 0.0, "last": None}


def _stat_to_dict(st: TrackContextStats):
    return {"play": st.play_count, "skip": st.skip_count,
            "manual_pick": st.manual_pick_count, "avg_completion": st.avg_completion,
            "last": st.last_played_at}


def build_queue(session: Session, *, scene: str, time_ctx: str, now: datetime,
                limit: int = 20, seed_track_id: int | None = None) -> list[Track]:
    tracks = session.scalars(select(Track).where(Track.status == "active")).all()
    stats = session.scalars(select(TrackContextStats)).all()
    by_track: dict[int, dict[str, dict]] = {}
    agg: dict[int, dict] = {}
    for st in stats:
        by_track.setdefault(st.track_id, {})[st.context_key] = _stat_to_dict(st)
        a = agg.setdefault(st.track_id, {"manual_pick": 0, "avg_completion": 0.0, "n": 0})
        a["manual_pick"] += st.manual_pick_count
        a["avg_completion"] += st.avg_completion
        a["n"] += 1

    seed_tags = set()
    if seed_track_id is not None:
        seed_tags = {
            (t.tag_type, t.tag_value)
            for t in session.scalars(select(TrackTag).where(TrackTag.track_id == seed_track_id))
        }

    is_rediscover = scene == REDISCOVER_SCENE
    weights = REDISCOVER_WEIGHTS if is_rediscover else WEIGHTS

    scored = []
    for t in tracks:
        ctxs = by_track.get(t.id, {})
        if is_rediscover:
            # Rank by global staleness across every context, not a single scene.
            global_last = max((c["last"] for c in ctxs.values() if c["last"]), default=None)
            scene_stat = {**_empty_stat(), "last": global_last}
        else:
            scene_stat = ctxs.get(scene, _empty_stat())
        time_stat = ctxs.get(time_ctx, _empty_stat())
        a = agg.get(t.id, {"manual_pick": 0, "avg_completion": 0.0, "n": 0})
        global_agg = {"manual_pick": a["manual_pick"],
                      "avg_completion": (a["avg_completion"] / a["n"]) if a["n"] else 0.0}
        seed_match = 0.0
        if seed_tags and t.id != seed_track_id:
            t_tags = {(tg.tag_type, tg.tag_value)
                      for tg in session.scalars(select(TrackTag).where(TrackTag.track_id == t.id))}
            if seed_tags & t_tags:
                seed_match = 1.0
        scored.append((score_track(now, scene_stat, time_stat, global_agg, seed_match, weights), t))

    scored.sort(key=lambda x: x[0], reverse=True)
    return [t for _, t in scored[:limit]]
```

```python
from datetime import datetime
from sqlalchemy import select
from sqlalchemy.orm import Session
from echo.models import PlayEvent, TrackContextStats
from echo.time_buckets import time_bucket

_SKIP_ACTIONS = {"skip", "too_noisy"}
_MANUAL_ACTIONS = {"manual_pick"}


def _upsert_stat(session: Session, track_id: int, ctx: str, completion: float,
                 action: str, now: datetime) -> None:
    st = session.scalar(
        select(TrackContextStats).where(
            TrackContextStats.track_id == track_id, TrackContextStats.context_key == ctx)
    )
    if st is None:
        st = TrackContextStats(track_id=track_id, context_key=ctx)
        session.add(st)
        st.play_count = 0
        st.skip_count = 0
        st.manual_pick_count = 0
        st.avg_completion = 0.0
    # running average of completion over play occurrences
    n = st.play_count
    st.avg_completion = (st.avg_completion * n + completion) / (n + 1)
    st.play_count = n + 1
    if action in _SKIP_ACTIONS:
        st.skip_count += 1
    if action in _MANUAL_ACTIONS:
        st.manual_pick_count += 1
    st.last_played_at = now


def record_event(session: Session, *, track_id: int, scene: str, action: str,
                 listened_seconds: int, duration_seconds: int, now: datetime,
                 source: str = "youtube") -> PlayEvent:
    completion = (listened_seconds / duration_seconds) if duration_seconds else 0.0
    completion = max(0.0, min(1.0, completion))
    ev = PlayEvent(track_id=track_id, played_at=now, scene=scene, action=action,
                   source=source, listened_seconds=listened_seconds,
                   completion_ratio=completion)
    session.add(ev)
    _upsert_stat(session, track_id, scene, completion, action, now)
    _upsert_stat(session, track_id, time_bucket(now.hour), completion, action, now)
    return ev
```

```python
@app.get("/api/queue", dependencies=[Depends(require_token)])
def queue(scene: str = Query(...), seed_video_id: str | None = None,
          session: Session = Depends(get_session)):
    now = datetime.now(timezone.utc)
    seed_id = None
    if seed_video_id:
        seed = session.scalar(select(Track).where(Track.youtube_video_id == seed_video_id))
        seed_id = seed.id if seed else None
    tracks = build_queue(session, scene=scene, time_ctx=time_bucket(now.hour),
                         now=now, limit=20, seed_track_id=seed_id)
    return {"scene": scene, "queue": [_track_json(t) for t in tracks]}


@app.get("/api/alerts", dependencies=[Depends(require_token)])
def alerts(session: Session = Depends(get_session)):
    rows = session.scalars(
        select(Alert).where(Alert.resolution_status == "open").order_by(Alert.detected_at.desc())
    ).all()
    return [{"id": a.id, "kind": a.kind, "source_track_id": a.source_track_id,
             "title_at_time": a.title_at_time, "candidate": a.candidate,
             "detected_at": a.detected_at.isoformat()} for a in rows]


@app.post("/api/feedback", dependencies=[Depends(require_token)])
def feedback(body: FeedbackIn, session: Session = Depends(get_session)):
    track = session.scalar(select(Track).where(Track.youtube_video_id == body.youtube_video_id))
    if track is None:
        raise HTTPException(status_code=404, detail="track not found")
    record_event(session, track_id=track.id, scene=body.scene, action=body.action,
                 listened_seconds=body.listened_seconds,
                 duration_seconds=body.duration_seconds, now=datetime.now(timezone.utc))
    session.commit()
    return {"ok": True}


@app.post("/api/alerts/{alert_id}/resolve", dependencies=[Depends(require_token)])
def resolve(alert_id: int, status: str = Query("resolved"),
            session: Session = Depends(get_session)):
    a = session.get(Alert, alert_id)
    if a is None:
        raise HTTPException(status_code=404, detail="alert not found")
    a.resolution_status = status
    session.commit()
    return {"ok": True}
```

```python
import logging
from datetime import datetime, timezone
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from echo.config import settings
from echo.db import SessionLocal
from echo.ingest.youtube import get_playlist_items, search_videos
from echo.registry import upsert_track
from echo.snapshot import run_watchdog

log = logging.getLogger("echo.scheduler")


def run_daily_snapshot() -> None:
    if not settings.playlist_id:
        log.warning("ECHO_PLAYLIST_ID not set; skipping snapshot")
        return
    session = SessionLocal()
    try:
        fetched = get_playlist_items(settings.playlist_id)  # raises on API failure
        # keep registry in sync with currently-present items
        for it in fetched:
            if it.available and it.source_track_id:
                upsert_track(session, video_id=it.source_track_id,
                             title=it.title_at_time, artist=it.artist_at_time)
        changes = run_watchdog(session, settings.playlist_id, fetched, search_fn=search_videos)
        session.commit()
        log.info("snapshot ok: %d missing, %d added, %d renamed",
                 len(changes.missing), len(changes.added), len(changes.renamed))
    except Exception:
        session.rollback()
        log.exception("daily snapshot failed; previous snapshot retained")
    finally:
        session.close()


def start_scheduler() -> BackgroundScheduler:
    sched = BackgroundScheduler(timezone="UTC")
    sched.add_job(run_daily_snapshot,
                  CronTrigger(hour=settings.snapshot_hour, minute=0),
                  id="daily_snapshot", replace_existing=True)
    sched.start()
    return sched
```
